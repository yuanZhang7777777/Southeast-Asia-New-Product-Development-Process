```js
'use strict';

require('dotenv').config();

const Util = require('@alicloud/tea-util');
const dingtalkcard_1_0 = require('@alicloud/dingtalk/card_1_0');
const OpenApi = require('@alicloud/openapi-client');

function createClient() {
  const config = new OpenApi.Config({});
  config.protocol = 'https';
  config.regionId = 'central';
  return new dingtalkcard_1_0.default(config);
}

function stringifyCardParamMap(obj) {
  const result = {};
  for (const [key, value] of Object.entries(obj)) {
    result[key] = typeof value === 'string' ? value : JSON.stringify(value ?? '');
  }
  return result;
}

async function sendCard({ accessToken, userId, outTrackId, cardParamMap }) {
  const client = createClient();
  const headers = new dingtalkcard_1_0.CreateAndDeliverHeaders({});
  headers.xAcsDingtalkAccessToken = accessToken;

  const robotCode = process.env.DINGTALK_ROBOT_CODE || process.env.DINGTALK_CLIENT_ID;
  const cardTemplateId = 'a4053068-a70c-4ec8-a5b7-293d38a67a84.schema';
  const lastMessage = cardParamMap.summary_text;

  const imRobotOpenDeliverModel = new dingtalkcard_1_0.CreateAndDeliverRequestImRobotOpenDeliverModel({
    spaceType: 'IM_ROBOT',
    robotCode,
  });

  const imRobotOpenSpaceModel = new dingtalkcard_1_0.CreateAndDeliverRequestImRobotOpenSpaceModel({
    supportForward: true,
    lastMessageI18n: { ZH_CN: lastMessage },
    searchSupport: new dingtalkcard_1_0.CreateAndDeliverRequestImRobotOpenSpaceModelSearchSupport({
      searchIcon: '',
      searchDesc: `${cardParamMap.card_title} ${lastMessage}`.slice(0, 200),
    }),
  });

  const cardData = new dingtalkcard_1_0.CreateAndDeliverRequestCardData({
    cardParamMap: stringifyCardParamMap(cardParamMap),
  });

  const request = new dingtalkcard_1_0.CreateAndDeliverRequest({
    userId,
    cardTemplateId,
    outTrackId,
    callbackType: 'STREAM',
    cardData,
    imRobotOpenSpaceModel,
    imRobotOpenDeliverModel,
    openSpaceId: `dtv1.card//im_robot.${userId}`,
    userIdType: 1,
  });

  return client.createAndDeliverWithOptions(request, headers, new Util.RuntimeOptions({}));
}

module.exports = { sendCard };
```
